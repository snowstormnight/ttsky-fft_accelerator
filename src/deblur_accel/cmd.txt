# Minimal full blur-removal flow (single hardware top: deblur.sv)
# image/clear input -> internal Gaussian blur -> deblur top(FFT+Wiener+Mult+IFFT) -> results/

# 1) Activate venv
# source src/fft/.venv/bin/activate

# 2) Enter folder
# cd src/deblur_accel

# 3) Put clear grayscale images in ./image

# 4) Run pipeline
python PY_deblur_pipeline.py \
  --images-dir image \
  --outdir results \
  --max-n 256 \
  --img-n 256 \
  --fft-logn 8 \
  --lanes 4 \
  --k-values 1,2,4,8,16,32,64,128,256,512,1024 \
  --h-scale 32767 \
  --psf-kernel 9 \
  --psf-sigma 2.0 \
  --deblur-psf-sigmas 1.2,1.6,2.0,2.4,2.8,3.2 \
  --data-w 24 \
  --frac-w 15 \
  --mult-frac 15 \
  --mult-saturate 1 \
  --pre-ifft-shifts 12 \
  --fft-to-ifft-maps 0 \
  --selection-metric sharpness

# Notes:
# - This uses PSF-bank search automatically (no manual PSF input required).
# - Set --selection-metric rmse only for synthetic tests where reference is known.

# Final files per case in results/<image>_N*_L*/ :
# - original.png
# - blurred.png
# - deblurred.png
# - metrics.txt
